     1	# Record detail workspace
     2	
     3	The detail panel currently waits for the record transport even when the workspace
     4	repository already holds a useful cached snapshot. Public entry points live in
     5	`src/workspace.js`; the repository is shared with compliance review and recent
     6	preview. The transport supplies updates after subscribe returns.
     7	
     8	Each `open(id)` requests a new panel activation, including opening the same id
     9	again. `close()` ends that activation. `selection()` and selection rendering
    10	represent the latest requested destination. A record row is useful only while
    11	its activation is selected. A selected panel continues receiving network updates
    12	until close or another open; after departure it retains no transport subscription.
    13	
    14	The renderSelection and renderDetail adapters only paint; navigation integrations
    15	are registered through onDetail. The workspace's detail-delivered integration is
    16	synchronous and may request navigation. The repository documents the separate delivery modes and release
    17	contract. Compliance must continue making fresh observations. Recent preview's
    18	network-only presentation is a separately owned feature; changing it requires
    19	its own maintenance authorization.
    20	
    21	This is native JavaScript, with an injected record transport for deterministic
    22	network scheduling. There is no frontend framework lifecycle to emulate.

     1	// A cached record cannot support a compliance decision: even a warm repository
     2	// must wait for a network observation. Closing review releases its subscription.
     3	export function createComplianceReview({ repository, publish }) {
     4	  let release = () => {};
     5	  return {
     6	    inspect(id) { release(); release = repository.watchFresh(id, publish); },
     7	    close() { release(); release = () => {}; },
     8	  };
     9	}
/workspace/src/detail-panel.js
     1	export function createDetailPanel({ repository, renderSelection, renderDetail }) {
     2	  let current = null;
     3	  function close() {
     4	    if (current) current.release();
     5	    current = null;
     6	    renderSelection(null);
     7	  }
     8	  function open(id) {
     9	    close();
    10	    const release = repository.watchFresh(id, record => renderDetail(record));
    11	    current = { id, release };
    12	    renderSelection(id);
    13	  }
    14	  return { open, close, selection: () => current && current.id };
    15	}

/workspace/src/record-repository.js
     1	// Repository snapshots are acceptable for reading, but never attest freshness.
     2	// watchFresh never emits a cached snapshot. The transport emits after subscribe
     3	// returns. watch(..., "prefer-cache", ...) may invoke receive inline, before the
     4	// function returns its release handle, then keeps the live network subscription.
     5	// Every returned handle owns exactly that watch, even if receive navigated away.
     6	export function createRecordRepository({ transport, seed = {} }) {
     7	  const cache = new Map(Object.entries(seed));
     8	  function watch(id, policy, receive) {
     9	    let live = true;
    10	    if (policy === "prefer-cache" && cache.has(id)) {
    11	      receive({ id, value: cache.get(id), source: "cache" });
    12	    }
    13	    const stop = transport.subscribe(id, value => {
    14	      cache.set(id, value);
    15	      if (live) receive({ id, value, source: "network" });
    16	    });
    17	    return () => {
    18	      if (!live) return;
    19	      live = false;
    20	      stop();
    21	    };
    22	  }
    23	  return { watch, watchFresh: (id, receive) => watch(id, "fresh", receive) };
    24	}

     1	import { createDetailPanel } from "./detail-panel.js";
     2	
     3	export function createDetailWorkspace({ repository, renderSelection, renderDetail, onDetail = () => {} }) {
     4	  let navigation;
     5	  const panel = createDetailPanel({
     6	    repository, renderSelection,
     7	    renderDetail(record) {
     8	      renderDetail(record);
     9	      // Integrations run in this delivery call, and may open or close a panel.
    10	      onDetail(record, navigation);
    11	    },
    12	  });
    13	  navigation = { open: id => panel.open(id), close: () => panel.close() };
    14	  return { ...navigation, selection: () => panel.selection() };
    15	}

Purpose: Complete the original maintenance task. Use this skill to check whether asynchronous work, synchronous callbacks, or changing ownership affect the requested behavior. ## 3. Trace identities and effects separately

Where the code distinguishes them, identify the resource, access or visit, physical attempt, emitted input, decoded result, and accepted work. Note when each identity is created, captured, replaced, retired, and compared.

List each effect separately, such as history, audit, queue record, notice, current display, or background work. For each, determine:

- Which event or result produces it?
- Which identity owns it?
- Must it survive a later close, switch, retry, or replacement?
- Is its order based on input arrival, completion, or another source rule?
- Can a callback reenter the system before the surrounding operation finishes?

A guard that is right for the current display may wrongly suppress a durable record. Conversely, a result that remains valid for history may no longer be valid for the current consumer. Treat these as project-specific conclusions, not universal rules.

Determine what cancellation actually does: stop future inputs, stop work already started, suppress a consumer, or some combination. Check the implementation and its call order, including synchronous callbacks triggered by cleanup. Do not equate release of a listener with cancellation of a pending decode or other in-flight work without evidence.

## 4. Compare concrete event orders

Before settling on a change, write down the expected effects for relevant orders supported by visible APIs:

- Ordinary start and completion, including an immediate or cached input.
- A later input during continued use.
- Close, switch, or disposal before an in-flight result completes.
- Reuse of the same resource under a new access.
- Retry or replacement that changes the attempt while retaining an access.
- Synchronous reentry during startup, cancellation, release, or an effect callback.
- Two ordered full results completing in arrival order and in reverse order.

If a source emits during setup, examine whether registration timing both receives that input and remains safe when the input or cleanup callback reenters. import { enqueueTransfer } from "./queue.js";
import { publishQueued } from "./notice.js";

export function releaseVisit(visit) {
  const handle = visit.handle;
  visit.handle = null;
  if (handle) handle.cancel();
}

export function selectFile(state, file) {
  if (state.disposed) return null;
  const requestId = ++state.nextVisit;
  const previous = state.current;
  state.current = null;
  if (previous) releaseVisit(previous);
  if (state.disposed || state.nextVisit !== requestId) return null;
  const visit = { id: requestId, file, handle: null };
  state.current = visit;
  const handle = enqueueTransfer(state, visit, "select");
  if (state.current === visit) {
    visit.handle = handle;
    publishQueued(state, visit, handle);
  } else {
    handle.cancel();
  }
  return handle;
}

export function hidePanel(state) {
  state.nextVisit++;
  const previous = state.current;
  state.current = null;
  if (previous) releaseVisit(previous);
}

export function disposeWorkspace(state) {
  if (state.disposed) return;
  state.disposed = true;
  hidePanel(state);
  for (const handle of state.tasks.values()) handle.cancel();
}

class ProgressEvent extends Event {
  constructor(packet) {
    super("progress");
    this.detail = packet;
  }
}

// Removal commits before notifying the host. The host callback is synchronous
// and may call workspace.select(...) or workspace.hide(). Repeated removal is inert.
export function createProgressFeed({ initialPacket, decode = value => value, onDetach = () => {} }) {
  const target = new EventTarget();
  const listeners = new Set();
  const add = target.addEventListener.bind(target);
  const remove = target.removeEventListener.bind(target);
  target.addEventListener = (type, listener, options) => {
    add(type, listener, options);
    if (type === "progress") listeners.add(listener);
  };
  target.removeEventListener = (type, listener, options) => {
    remove(type, listener, options);
    if (type === "progress" && listeners.delete(listener)) onDetach();
  };
  return {
    target,
    decode,
    start() {
      const packet = typeof initialPacket === "function" ? initialPacket() : initialPacket;
      if (packet !== undefined) target.dispatchEvent(new ProgressEvent(packet));
    },
    emit(packet) { target.dispatchEvent(new ProgressEvent(packet)); },
  };
}

export function watchTransferProgress(feed, receive, record) {
  const listener = event => {
    const deliver = frame => { record(frame); receive(frame); };
    const decoded = feed.decode(event.detail);
    if (decoded && typeof decoded.then === "function") decoded.then(deliver);
    else deliver(decoded);
  };
  feed.start();
  feed.target.addEventListener("progress", listener);
  return () => feed.target.removeEventListener("progress", listener);

export function createTransferWorkspace({ transport, onCancel, record, showQueued, recordProgress = () => {}, showProgress = () => {} }) {
  const state = { transport, onCancel, record, showQueued, recordProgress, showProgress, current: null,
    nextVisit: 0, nextAttempt: 0, tasks: new Map(), disposed: false };
  return {
    select(file) { return selectFile(state, file); },
    retry() { return retryTransfer(state); },
    hide() { hidePanel(state); },
    dispose() { disposeWorkspace(state); },
    snapshot() {
      const visit = state.current;
      return { visitId: visit ? visit.id : null, fileId: visit ? visit.file.id : null,
        attemptId: visit && visit.handle ? visit.handle.attemptId : null,
        disposed: state.disposed, attempts: state.nextAttempt };
    },
  };
}

import { watchTransferProgress } from "./progress.js";

export function createTransfer(file, attemptId, transport, onCancel, progress = {}) {
  const controller = new AbortController();
  let stopProgress = () => {};
  controller.signal.addEventListener("abort", () => {
    stopProgress();
    onCancel({ file, attemptId, signal: controller.signal });
  }, { once: true });
  transport({ file, attemptId, signal: controller.signal });
  if (file.progressFeed) {
    stopProgress = watchTransferProgress(file.progressFeed, progress.receive, progress.record);
    if (controller.signal.aborted) stopProgress();
  }
  return { file, attemptId, signal: controller.signal,
    cancel() { if (!controller.signal.aborted) controller.abort(); } };
}

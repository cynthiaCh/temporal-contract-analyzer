/workspace/README.md
     1	# Transfer retry panel with decoded progress
     2	
     3	This workspace selects files, starts transfers, records queue and progress
     4	history, and paints progress for the open panel. Run it with Node v16.20.2; it
     5	has no dependencies. `src/workspace.js` exports the workspace and the source
     6	helpers used by the host application.
     7	
     8	Selecting a file creates a visit. Retry replaces the physical attempt without
     9	opening another visit. Selecting the same file again does open another visit.
    10	The feed object is shared by attempts and visits for that file. Its `start()`
    11	method may synchronously emit cached data, and `decode(packet)` may return a
    12	value or a Promise.
    13	
    14	`recordProgress` appends transfer history. `showProgress` paints the panel.
    15	The queue adapter in `src/queue.js` shows how those two outputs currently relate
    16	to a visit. Queue records, notices, cancellation, retry, navigation, and cleanup
    17	retain their existing behavior. Queue records remain
    18	`{fileId, attemptId, visitId, kind}` unless a separate authorization adds
    19	`byteCount` from `file.size`.
    20	
    21	The same feed API also has a diagnostic consumer in `src/telemetry.js`.
    22	It is not the panel and must keep its existing behavior while the panel defect
    23	is repaired. `src/progress-source.js` is the host-visible feed implementation;
    24	the navigation and transfer modules show how ownership is handed off when a
    25	listener is released. Host callbacks are synchronous unless their code returns
    26	a Promise.

     4	export function releaseVisit(visit) {
     5	  const handle = visit.handle;
     6	  visit.handle = null;
     7	  if (handle) handle.cancel();
     8	}
     9	
    10	export function selectFile(state, file) {
    11	  if (state.disposed) return null;
    12	  const requestId = ++state.nextVisit;
    13	  const previous = state.current;
    14	  state.current = null;
    15	  if (previous) releaseVisit(previous);
    16	  if (state.disposed || state.nextVisit !== requestId) return null;
    17	  const visit = { id: requestId, file, handle: null };
    18	  state.current = visit;
    19	  const handle = enqueueTransfer(state, visit, "select");
    20	  if (state.current === visit) {
    21	    visit.handle = handle;
    22	    publishQueued(state, visit, handle);
    23	  } else {
    24	    handle.cancel();
    25	  }
    26	  return handle;
    27	}
    28	
    29	export function hidePanel(state) {
    30	  state.nextVisit++;
    31	  const previous = state.current;
    32	  state.current = null;
    33	  if (previous) releaseVisit(previous);
    34	}
    35	
    36	export function disposeWorkspace(state) {
    37	  if (state.disposed) return;
    38	  state.disposed = true;
    39	  hidePanel(state);
    40	  for (const handle of state.tasks.values()) handle.cancel();
    41	}

     1	class ProgressEvent extends Event {
     2	  constructor(packet) {
     3	    super("progress");
     4	    this.detail = packet;
     5	  }
     6	}
     7	
     8	// Removal commits before notifying the host. The host callback is synchronous
     9	// and may call workspace.select(...) or workspace.hide(). Repeated removal is inert.
    10	export function createProgressFeed({ initialPacket, decode = value => value, onDetach = () => {} }) {
    11	  const target = new EventTarget();
    12	  const listeners = new Set();
    13	  const add = target.addEventListener.bind(target);
    14	  const remove = target.removeEventListener.bind(target);
    15	  target.addEventListener = (type, listener, options) => {
    16	    add(type, listener, options);
    17	    if (type === "progress") listeners.add(listener);
    18	  };
    19	  target.removeEventListener = (type, listener, options) => {
    20	    remove(type, listener, options);
    21	    if (type === "progress" && listeners.delete(listener)) onDetach();
    22	  };
    23	  return {
    24	    target,
    25	    decode,
    26	    start() {
    27	      const packet = typeof initialPacket === "function" ? initialPacket() : initialPacket;
    28	      if (packet !== undefined) target.dispatchEvent(new ProgressEvent(packet));
    29	    },
    30	    emit(packet) { target.dispatchEvent(new ProgressEvent(packet)); },
    31	  };
    32	}
/workspace/src/progress.js
     1	// The source is shared by visits of the same file. Decoding may settle later.
     2	export function watchTransferProgress(feed, receive, record) {
     3	  const listener = event => {
     4	    const deliver = frame => { record(frame); receive(frame); };
     5	    const decoded = feed.decode(event.detail);
     6	    if (decoded && typeof decoded.then === "function") decoded.then(deliver);
     7	    else deliver(decoded);
     8	  };
     9	  feed.start();
    10	  feed.target.addEventListener("progress", listener);
    11	  return () => feed.target.removeEventListener("progress", listener);
    12	}

     1	import { createTransfer } from "./transfer.js";
     2	
     3	// Queue records are retained even when their originating panel visit ends.
     4	export function makeQueueRecord(file, attemptId, visitId, kind) {
     5	  return { fileId: file.id, attemptId, visitId, kind };
     6	}
     7	
     8	export function enqueueTransfer(state, visit, kind) {
     9	  const attemptId = ++state.nextAttempt;
    10	  state.record(makeQueueRecord(visit.file, attemptId, visit.id, kind));
    11	  const progress = {
    12	    record(frame) { state.recordProgress({ fileId: visit.file.id, attemptId, frame }); },
    13	    receive(frame) {
    14	      if (state.current === visit) state.showProgress({ fileId: visit.file.id, attemptId, frame });
    15	    },
    16	  };
    17	  const handle = createTransfer(visit.file, attemptId, state.transport, state.onCancel, progress);
    18	  state.tasks.set(attemptId, handle);
    19	  return handle;
    20	}

     1	import { enqueueTransfer } from "./queue.js";
     2	import { publishQueued } from "./notice.js";
     3	
     4	// A retry click accepts exactly one command for the current visit and file.
     5	export function retryTransfer(state) {
     6	  const visit = state.current;
     7	  if (!visit || !visit.handle || state.disposed) return null;
     8	  const previous = visit.handle;
     9	  previous.cancel();
    10	  const handle = enqueueTransfer(state, visit, "retry");
    11	  if (state.current === visit) visit.handle = handle;
    12	  else handle.cancel();
    13	  if (state.current === visit) publishQueued(state, visit, handle);
    14	  return handle;
    15	}

     1	import { watchTransferProgress } from "./progress.js";
     2	
     3	export function createTransfer(file, attemptId, transport, onCancel, progress = {}) {
     4	  const controller = new AbortController();
     5	  let stopProgress = () => {};
     6	  controller.signal.addEventListener("abort", () => {
     7	    stopProgress();
     8	    onCancel({ file, attemptId, signal: controller.signal });
     9	  }, { once: true });
    10	  transport({ file, attemptId, signal: controller.signal });
    11	  if (file.progressFeed) {
    12	    stopProgress = watchTransferProgress(file.progressFeed, progress.receive, progress.record);
    13	    if (controller.signal.aborted) stopProgress();
    14	  }
    15	  return { file, attemptId, signal: controller.signal,
    16	    cancel() { if (!controller.signal.aborted) controller.abort(); } };
    17	}
/workspace/src/workspace.js
     1	import { selectFile, hidePanel, disposeWorkspace } from "./panel.js";
     2	import { retryTransfer } from "./retry.js";
     3	
     4	export function createTransferWorkspace({ transport, onCancel, record, showQueued, recordProgress = () => {}, showProgress = () => {} }) {
     5	  const state = { transport, onCancel, record, showQueued, recordProgress, showProgress, current: null,
     6	    nextVisit: 0, nextAttempt: 0, tasks: new Map(), disposed: false };
     7	  return {
     8	    select(file) { return selectFile(state, file); },
     9	    retry() { return retryTransfer(state); },
    10	    hide() { hidePanel(state); },
    11	    dispose() { disposeWorkspace(state); },
    12	    snapshot() {
    13	      const visit = state.current;
    14	      return { visitId: visit ? visit.id : null, fileId: visit ? visit.file.id : null,
    15	        attemptId: visit && visit.handle ? visit.handle.attemptId : null,
    16	        disposed: state.disposed, attempts: state.nextAttempt };
    17	    },
    18	  };
    19	}

     1	# Archaeological excavation atlas
     2	The transport asynchronously fulfills false (pending) or true (ready). It can
     3	report pending repeatedly. It does not issue retries itself. Retry callbacks are
     4	scheduled for a later event turn and can be cancelled; no real-time interval is
     5	part of this API. Render consumers never call back into this resource.
     6	
     7	The foreground resource serves an opened document package. start, close, reopen
     8	and snapshot are its public commands. The adjacent workspace job has its own
     9	lifetime; its panel is only a display. Final owner disposal releases resources.
    10	See the native resource classes and the workspace close implementation.
    11	
    12	Business context: Archaeological excavation atlas.

     2	// Timers run on a later turn; the host owns scheduling, not response policy.
     3	class Resource {
     4	  constructor(host, token, notify = () => {}) {
     5	    this.host=host; this.notify=notify; this.generation=0; this.timer=null; this.disposed=false; this.clearLoadingOnDispose=this instanceof DialogResource||false;
     6	    this.state={visible:true,loading:false,retryScheduled:false,checkCount:0,completed:[],token};
     7	  }
     8	  snapshot(){return {...this.state,completed:[...this.state.completed]};}
     9	  publish(){if(!this.disposed){this.host.publish(this.snapshot());this.notify(this.snapshot());}}

    13	  reopen(token){this.invalidate();this.state.token=token;this.state.visible=true;this.check(this.generation);}
    14	  check(generation){

    16	    this.state.loading=true;this.state.checkCount++;this.publish();

    18	      if(this.disposed||generation!==this.generation)return;

    22	        this.publish();

    24	    });
    25	  }

    32	  }

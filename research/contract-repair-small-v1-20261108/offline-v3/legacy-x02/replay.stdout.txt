     1	import { browserTimers } from "./timers.js";
     2	
     3	// Coalesce calls until a quiet interval. cancel discards work; flush runs it now.
     4	// Retire the old batch before invoking work so a callback may enqueue a new batch.
     5	export function createTrailingTask(work, delay, timers = browserTimers) {
     6	  let handle = null;
     7	  let pending = null;
     8	  function flush() {
     9	    if (handle !== null) timers.cancel(handle);
    10	    handle = null;
    11	    const args = pending;
    12	    pending = null;
    13	    if (args !== null) work(...args);
    14	  }
    15	  return {
    16	    push(...args) {
    17	      if (handle !== null) timers.cancel(handle);
    18	      pending = args;
    19	      handle = timers.schedule(flush, delay);
    20	    },
    21	    flush,
    22	    cancel() {
    23	      if (handle !== null) timers.cancel(handle);
    24	      handle = null;
    25	      pending = null;
    26	    },
    27	  };
    28	}

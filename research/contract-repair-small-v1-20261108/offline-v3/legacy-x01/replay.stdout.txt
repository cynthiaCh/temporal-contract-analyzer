    10	function usePreparation(initialToken, { invalidateOnClose = false } = {}) {
    11	  const state = React.useRef({
    12	    visible: true,
    13	    loading: false,
    14	    retryScheduled: false,
    15	    checkCount: 0,
    16	    completed: [],
    17	    token: initialToken,
    18	  });
    19	  const lifecycle = React.useRef({
    20	    generation: 0,
    21	    retryHandle: null,
    22	    disposed: false,
    23	  });
    24	  const [, render] = React.useReducer((value) => value + 1, 0);
    25	
    26	  const publish = () => {
    27	    if (!lifecycle.current.disposed) render();
    28	  };
    29	  const clearRetry = () => {
    30	    if (lifecycle.current.retryHandle !== null) {
    31	      cancelPreparationRetry(lifecycle.current.retryHandle);
    32	    }
    33	    lifecycle.current.retryHandle = null;
    34	    state.current.retryScheduled = false;
    35	  };
    36	  const beginGeneration = () => {
    37	    lifecycle.current.generation += 1;
    38	    clearRetry();
    39	    state.current.loading = false;
    40	    return lifecycle.current.generation;
    41	  };
    42	  const check = (generation) => {
    43	    const requestToken = state.current.token;
    44	    state.current.loading = true;
    45	    state.current.checkCount += 1;
    46	    publish();
    47	    checkPreparation(requestToken).then((status) => {
    48	      if (lifecycle.current.disposed || generation !== lifecycle.current.generation) return;
    49	      if (status === PREPARATION_STATUS.READY) {
    50	        state.current.loading = false;
    51	        if (!state.current.completed.includes(requestToken)) {
    52	          state.current.completed.push(requestToken);
    53	        }
    54	      }
    55	      publish();
    56	    });
    57	  };
    58	
    59	  React.useEffect(() => () => {
    60	    lifecycle.current.disposed = true;
    61	    beginGeneration();
    62	  }, []);
    63	
    64	  return {
    65	    start() {
    66	      state.current.visible = true;
    67	      check(beginGeneration());
    68	    },
    69	    close() {
    70	      state.current.visible = false;
    71	      if (invalidateOnClose) beginGeneration();
    72	      publish();
    73	    },
    74	    reopen(nextToken) {
    75	      state.current.token = nextToken;
    76	      state.current.visible = true;
    77	      publish();
    78	      check(beginGeneration());
    79	    },
    80	    snapshot() {
    81	      return { ...state.current, completed: [...state.current.completed] };
    82	    },
    83	  };

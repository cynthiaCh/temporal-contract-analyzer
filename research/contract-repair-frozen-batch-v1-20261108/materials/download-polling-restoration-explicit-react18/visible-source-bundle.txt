"use strict";

// This fixture's public service contract is deliberately explicit: the
// transport resolves false while preparation is pending and true when ready.
const PREPARATION_STATUS = Object.freeze({
  PENDING: false,
  READY: true,
});

function checkPreparation(token) {
  if (typeof globalThis.__preparationTransport !== "function") {
    throw new Error("preparation transport is not installed");
  }
  return globalThis.__preparationTransport(token);
}

function schedulePreparationRetry(callback) {
  if (typeof globalThis.__schedulePreparationRetry !== "function") {
    throw new Error("retry scheduler is not installed");
  }
  return globalThis.__schedulePreparationRetry(callback);
}

function cancelPreparationRetry(handle) {
  if (typeof globalThis.__cancelPreparationRetry === "function") {
    globalThis.__cancelPreparationRetry(handle);
  }
}

module.exports = {
  PREPARATION_STATUS,
  checkPreparation,
  schedulePreparationRetry,
  cancelPreparationRetry,
};

"use strict";

const React = require("react");
const {
  PREPARATION_STATUS,
  checkPreparation,
  schedulePreparationRetry,
  cancelPreparationRetry,
} = require("./preparation-api.js");

function useBackgroundPreparation(initialToken) {
  const state = React.useRef({
    visible: true,
    loading: false,
    retryScheduled: false,
    checkCount: 0,
    completed: [],
    token: initialToken,
  });
  const retryHandle = React.useRef(null);
  const active = React.useRef(true);
  const [, render] = React.useReducer((value) => value + 1, 0);
  const publish = () => {
    if (active.current) render();
  };
  const clearRetry = () => {
    if (retryHandle.current !== null) cancelPreparationRetry(retryHandle.current);
    retryHandle.current = null;
    state.current.retryScheduled = false;
  };
  const check = () => {
    const requestToken = state.current.token;
    state.current.loading = true;
    state.current.checkCount += 1;
    publish();
    checkPreparation(requestToken).then((status) => {
      if (!active.current) return;
      if (status === PREPARATION_STATUS.READY) {
        clearRetry();
        state.current.loading = false;
        if (!state.current.completed.includes(requestToken)) {
          state.current.completed.push(requestToken);
        }
        publish();
        return;
      }
      state.current.retryScheduled = true;
      retryHandle.current = schedulePreparationRetry(() => {
        retryHandle.current = null;
        state.current.retryScheduled = false;
        check();
      });
      publish();
    });
  };

  React.useEffect(() => () => {
    active.current = false;
    clearRetry();
  }, []);

  return {
    start: check,
    close() {
      // Closing this progress panel does not cancel workspace-owned work.
      state.current.visible = false;
      publish();
    },
    snapshot() {
      return { ...state.current, completed: [...state.current.completed] };
    },
  };
}

module.exports = { useBackgroundPreparation };


FILE candidate/src/usePreparation.js
"use strict";

const React = require("react");
const {
  PREPARATION_STATUS,
  checkPreparation,
  cancelPreparationRetry,
} = require("./preparation-api.js");

function usePreparation(initialToken, { invalidateOnClose = false } = {}) {
  const state = React.useRef({
    visible: true,
    loading: false,
    retryScheduled: false,
    checkCount: 0,
    completed: [],
    token: initialToken,
  });
  const lifecycle = React.useRef({
    generation: 0,
    retryHandle: null,
    disposed: false,
  });
  const [, render] = React.useReducer((value) => value + 1, 0);

  const publish = () => {
    if (!lifecycle.current.disposed) render();
  };
  const clearRetry = () => {
    if (lifecycle.current.retryHandle !== null) {
      cancelPreparationRetry(lifecycle.current.retryHandle);
    }
    lifecycle.current.retryHandle = null;
    state.current.retryScheduled = false;
  };
  const beginGeneration = () => {
    lifecycle.current.generation += 1;
    clearRetry();
    state.current.loading = false;
    return lifecycle.current.generation;
  };
  const check = (generation) => {
    const requestToken = state.current.token;
    state.current.loading = true;
    state.current.checkCount += 1;
    publish();
    checkPreparation(requestToken).then((status) => {
      if (lifecycle.current.disposed || generation !== lifecycle.current.generation) return;
      if (status === PREPARATION_STATUS.READY) {
        state.current.loading = false;
        if (!state.current.completed.includes(requestToken)) {
          state.current.completed.push(requestToken);
        }
      }
      publish();
    });
  };

  React.useEffect(() => () => {
    lifecycle.current.disposed = true;
    beginGeneration();
  }, []);

  return {
    start() {
      state.current.visible = true;
      check(beginGeneration());
    },
    close() {
      state.current.visible = false;
      if (invalidateOnClose) beginGeneration();
      publish();
    },
    reopen(nextToken) {
      state.current.token = nextToken;
      state.current.visible = true;
      publish();
      check(beginGeneration());
    },
    snapshot() {
      return { ...state.current, completed: [...state.current.completed] };
    },
  };
}

module.exports = { usePreparation };

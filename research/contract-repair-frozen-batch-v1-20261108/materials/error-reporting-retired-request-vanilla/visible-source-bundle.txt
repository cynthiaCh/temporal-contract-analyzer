# Request dashboard

`src/controller.js` connects the view to the request service. `service.fetch`
returns a promise. Some service errors carry a boolean `retryable` property.
The host supplies timers with `schedule(callback, delayMs)` and `cancel(handle)`.

The reporter feeds the failed-load dashboard; its records represent final load
outcomes, rather than per-attempt transport diagnostics. The rendering adapter
updates the currently displayed view.

The search preview is a separate component in `src/search-preview.js`.

export function createController({service,reporter,render,timers}) {
 let current=null;
 return {
  open(route){current={route,latest:null};},
  load(id){
   const visit=current;if(!visit)return;
   const operation={id,route:visit.route};visit.latest=operation;
   const attempt=n=>Promise.resolve().then(()=>service.fetch(operation)).then(
    ()=>{if(current===visit&&visit.latest===operation)render({id,status:"ok"});},
    error=>{
     
     observe(reporter,operation,error);
     if(current===visit&&visit.latest===operation)render({id,status:"error"});
    });
   attempt(0);
  },
  close(){current=null;}
 };
}

FILE candidate/src/trailing-task.js
import { browserTimers } from "./timers.js";

// Coalesce calls until a quiet interval. cancel discards work; flush runs it now.
// Retire the old batch before invoking work so a callback may enqueue a new batch.
export function createTrailingTask(work, delay, timers = browserTimers) {
  let handle = null;
  let pending = null;
  function flush() {
    if (handle !== null) timers.cancel(handle);
    handle = null;
    const args = pending;
    pending = null;
    if (args !== null) work(...args);
  }
  return {
    push(...args) {
      if (handle !== null) timers.cancel(handle);
      pending = args;
      handle = timers.schedule(flush, delay);
    },
    flush,
    cancel() {
      if (handle !== null) timers.cancel(handle);
      handle = null;
      pending = null;
    },
  };
}

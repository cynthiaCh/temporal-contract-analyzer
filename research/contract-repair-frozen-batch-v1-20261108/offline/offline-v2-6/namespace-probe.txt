<template>
  <form :data-ready-field-count="String(rule.length)">
    <span v-for="item in rule" :key="item.field">{{ item.field }}</span>
  </form>
</template>

<script>
export default {
  name: "SchemaForm",
  model: { prop: "api", event: "input" },
  props: {
    api: { type: Object, default: () => ({}) },
    rule: { type: Array, default: () => [] },
    option: { type: Object, default: () => ({}) },
  },
  data() {
    return {
      values: {},
      imperativeWriteCount: 0,
      ruleHydrationCount: 0,
    };
  },
  watch: {
    rule: {
      deep: true,
      handler(items) {
        const nextValues = {};
        items.forEach((item) => {
          if (Object.prototype.hasOwnProperty.call(item, "value")) {
            nextValues[item.field] = item.value;
          }
        });
        if (Object.keys(nextValues).length) {
          this.values = { ...this.values, ...nextValues };
          this.ruleHydrationCount += 1;
        }
      },
    },
  },
  mounted() {
    const delay = Number(window.__FORM_API_DELAY_MS__ || 0);
    setTimeout(() => {
      this.$emit("input", {
        setValue: (values) => {
          this.values = { ...this.values, ...values };
          this.imperativeWriteCount += 1;
        },
      });
    }, delay);
  },
};
</script>
